# merchant_db schema — Relationship Manager (RM) / signup tables (different database, same server)
- wbx_merchants(id, merchant_id)                                      # merchant_db; merchant_id -> webxpay_master.tbl_store.store_id ; id <- wbx_merchant_signups.merchant_id
- wbx_merchant_signups(id, merchant_id, live_rm_id)                   # merchant_db; merchant_id -> wbx_merchants.id (NOT store_id); live_rm_id -> wbx_live_rms.id
- wbx_live_rms(id, admin_user_id)                                     # merchant_db; id <- wbx_merchant_signups.live_rm_id
- wbx_admin_users(id, name)                                           # merchant_db; id <- wbx_live_rms.admin_user_id; name = RM name


# Database Relationships

- tbl_order.payment_id -> tbl_payment.payment_id
- tbl_order.store_id -> tbl_store.store_id
- tbl_order.payment_status_id -> tbl_payment_status.payment_status_id
- tbl_order.store_payment_gateway_id -> tbl_store_payment_gateway_2.store_payment_gateway_id
- tbl_store_payment_gateway_2.payment_gateway_id -> tbl_payment_gateway.payment_gateway_id
- tbl_payment_gateway.ipg_provider_id -> tbl_ipg_provider.ipg_provider_id
- tbl_order.payment_currency_id -> tbl_currency.currency_id
- tbl_order.processing_currency_id -> tbl_currency.currency_id
- tbl_pos_transactions.store_id -> tbl_store.store_id
- tbl_pos_transactions.bank_merchant_mid -> tbl_pos_store_bank_mid.bank_merchant_mid
- tbl_payment_gateway.currency_id -> tbl_currency.currency_id

# RM (Relationship Manager) name for a store — join chain across merchant_db (via wbx_merchants bridge):
- webxpay_master.tbl_store.store_id = merchant_db.wbx_merchants.merchant_id
- merchant_db.wbx_merchants.id = merchant_db.wbx_merchant_signups.merchant_id   (signup.merchant_id is the wbx_merchants PK, NOT the store_id)
- merchant_db.wbx_merchant_signups.live_rm_id -> merchant_db.wbx_live_rms.id
- merchant_db.wbx_live_rms.admin_user_id -> merchant_db.wbx_admin_users.id
- merchant_db.wbx_admin_users.name = the RM name
